// C Linked List Insertion in O(1)
typedef struct Node {
    int data;
    struct Node* next;
} Node;

void insertAtFront(Node** head, int new_data) {
    Node* new_node = (Node*)malloc(sizeof(Node));
    new_node->data = new_data;
    new_node->next = (*head);
    (*head) = new_node;
}
